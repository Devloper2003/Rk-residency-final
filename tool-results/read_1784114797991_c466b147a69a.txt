     1→"use client";
     2→
     3→import { useEffect, useState, useMemo } from "react";
     4→import { motion, AnimatePresence } from "framer-motion";
     5→import {
     6→  X, Calendar, Users, Check, Loader2, ChevronRight, ChevronLeft,
     7→  ShieldCheck, Sparkles, Mail, Phone, User, MessageSquare,
     8→} from "lucide-react";
     9→import { Button } from "@/components/ui/button";
    10→import { Input } from "@/components/ui/input";
    11→import { Label } from "@/components/ui/label";
    12→import {
    13→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    14→} from "@/components/ui/select";
    15→import { Calendar as CalendarPicker } from "@/components/ui/calendar";
    16→import { Diya, Lotus } from "./Motifs";
    17→import type { Room } from "./Rooms";
    18→import { toast } from "sonner";
    19→
    20→type Props = {
    21→  open: boolean;
    22→  onOpenChange: (o: boolean) => void;
    23→  preselectRoom?: Room | null;
    24→};
    25→
    26→const STEPS = ["Dates & Room", "Guests", "Payment", "Confirmation"] as const;
    27→type Step = (typeof STEPS)[number];
    28→
    29→const GST_RATE = 0.12; // 12% GST on hotels < ₹7,500
    30→const SERVICE_FEE_FLAT = 250;
    31→
    32→export function BookingWidget({ open, onOpenChange, preselectRoom }: Props) {
    33→  const [rooms, setRooms] = useState<Room[]>([]);
    34→  const [step, setStep] = useState<0 | 1 | 2 | 3>(0);
    35→  const [dateRange, setDateRange] = useState<{ from?: Date; to?: Date }>({});
    36→  const [selectedRoomId, setSelectedRoomId] = useState<string>("");
    37→  const [guests, setGuests] = useState({ adults: 2, children: 0 });
    38→  const [contact, setContact] = useState({ name: "", email: "", phone: "", requests: "" });
    39→  const [paymentMethod, setPaymentMethod] = useState<"RAZORPAY" | "STRIPE" | "PAY_AT_HOTEL">("RAZORPAY");
    40→  const [submitting, setSubmitting] = useState(false);
    41→  const [confirmedRef, setConfirmedRef] = useState<string | null>(null);
    42→
    43→  // Load rooms
    44→  useEffect(() => {
    45→    if (!open) return;
    46→    fetch("/api/rooms")
    47→      .then((r) => r.json())
    48→      .then((data) => setRooms(data.rooms || []))
    49→      .catch(() => {});
    50→  }, [open]);
    51→
    52→  // Preselect room when opening with one
    53→  useEffect(() => {
    54→    if (preselectRoom) {
    55→      setSelectedRoomId(preselectRoom.id);
    56→    }
    57→  }, [preselectRoom]);
    58→
    59→  // Reset on close
    60→  useEffect(() => {
    61→    if (!open) {
    62→      const t = setTimeout(() => {
    63→        setStep(0);
    64→        setDateRange({});
    65→        setSelectedRoomId("");
    66→        setGuests({ adults: 2, children: 0 });
    67→        setContact({ name: "", email: "", phone: "", requests: "" });
    68→        setConfirmedRef(null);
    69→      }, 300);
    70→      return () => clearTimeout(t);
    71→    }
    72→  }, [open]);
    73→
    74→  const selectedRoom = useMemo(
    75→    () => rooms.find((r) => r.id === selectedRoomId),
    76→    [rooms, selectedRoomId]
    77→  );
    78→
    79→  const nights = useMemo(() => {
    80→    if (!dateRange.from || !dateRange.to) return 0;
    81→    return Math.max(
    82→      0,
    83→      Math.round((dateRange.to.getTime() - dateRange.from.getTime()) / (1000 * 60 * 60 * 24))
    84→    );
    85→  }, [dateRange]);
    86→
    87→  const pricing = useMemo(() => {
    88→    if (!selectedRoom || nights === 0) return null;
    89→    const subtotal = selectedRoom.basePrice * nights;
    90→    const taxes = Math.round(subtotal * GST_RATE);
    91→    const serviceFee = SERVICE_FEE_FLAT * nights;
    92→    const total = subtotal + taxes + serviceFee;
    93→    return { subtotal, taxes, serviceFee, total, perNight: selectedRoom.basePrice };
    94→  }, [selectedRoom, nights]);
    95→
    96→  const canProceedDates = !!dateRange.from && !!dateRange.to && !!selectedRoomId && nights > 0;
    97→  const canProceedGuests =
    98→    contact.name.length > 1 &&
    99→    /^\S+@\S+\.\S+$/.test(contact.email) &&
   100→    contact.phone.length >= 8;
   101→
   102→  const handleConfirm = async () => {
   103→    if (!selectedRoom || !dateRange.from || !dateRange.to) return;
   104→    setSubmitting(true);
   105→    try {
   106→      const res = await fetch("/api/bookings", {
   107→        method: "POST",
   108→        headers: { "Content-Type": "application/json" },
   109→        body: JSON.stringify({
   110→          roomId: selectedRoom.id,
   111→          checkIn: dateRange.from.toISOString(),
   112→          checkOut: dateRange.to.toISOString(),
   113→          nights,
   114→          adults: guests.adults,
   115→          children: guests.children,
   116→          guestName: contact.name,
   117→          guestEmail: contact.email,
   118→          guestPhone: contact.phone,
   119→          specialRequests: contact.requests,
   120→          paymentMethod,
   121→        }),
   122→      });
   123→      const data = await res.json();
   124→      if (!res.ok) throw new Error(data.error || "Booking failed");
   125→      setConfirmedRef(data.referenceCode);
   126→      setStep(3);
   127→      toast.success("Booking confirmed!", {
   128→        description: `Reference ${data.referenceCode}`,
   129→      });
   130→    } catch (e: unknown) {
   131→      const msg = e instanceof Error ? e.message : "Please try again";
   132→      toast.error("Booking failed", { description: msg });
   133→    } finally {
   134→      setSubmitting(false);
   135→    }
   136→  };
   137→
   138→  return (
   139→    <AnimatePresence>
   140→      {open && (
   141→        <motion.div
   142→          initial={{ opacity: 0 }}
   143→          animate={{ opacity: 1 }}
   144→          exit={{ opacity: 0 }}
   145→          className="fixed inset-0 z-[70] flex items-end justify-center sm:items-center"
   146→          role="dialog"
   147→          aria-modal="true"
   148→          aria-label="Booking widget"
   149→        >
   150→          <div
   151→            className="absolute inset-0 bg-charcoal/70 backdrop-blur-sm"
   152→            onClick={() => onOpenChange(false)}
   153→          />
   154→          <motion.div
   155→            initial={{ y: "100%", opacity: 0.6 }}
   156→            animate={{ y: 0, opacity: 1 }}
   157→            exit={{ y: "100%", opacity: 0.6 }}
   158→            transition={{ duration: 0.45, ease: [0.16, 1, 0.3, 1] }}
   159→            className="relative flex max-h-[92vh] w-full max-w-3xl flex-col overflow-hidden rounded-t-3xl bg-ivory shadow-2xl sm:rounded-3xl"
   160→          >
   161→            {/* Header */}
   162→            <div className="flex items-center justify-between border-b border-charcoal/10 bg-teal px-5 py-4 text-ivory sm:px-6">
   163→              <div className="flex items-center gap-3">
   164→                <Lotus size={20} className="text-gold-soft" />
   165→                <div>
   166→                  <h3 className="font-serif text-lg font-semibold">
   167→                    Reserve your stay
   168→                  </h3>
   169→                  <p className="font-display text-xs uppercase tracking-wider text-ivory/70">
   170→                    Direct booking · Best price · Free cancellation
   171→                  </p>
   172→                </div>
   173→              </div>
   174→              <button
   175→                onClick={() => onOpenChange(false)}
   176→                className="grid h-9 w-9 place-items-center rounded-full border border-ivory/25 text-ivory transition-colors hover:bg-ivory/10 focus-ring"
   177→                aria-label="Close booking widget"
   178→              >
   179→                <X className="h-5 w-5" />
   180→              </button>
   181→            </div>
   182→
   183→            {/* Stepper */}
   184→            <div className="flex items-center justify-between gap-2 border-b border-charcoal/10 bg-ivory px-5 py-3 sm:px-6">
   185→              {STEPS.map((label, i) => (
   186→                <div key={label} className="flex flex-1 items-center gap-2">
   187→                  <div
   188→                    className={`grid h-7 w-7 shrink-0 place-items-center rounded-full border text-xs font-bold transition-all ${
   189→                      i < step
   190→                        ? "border-teal bg-teal text-ivory"
   191→                        : i === step
   192→                        ? "border-gold bg-gold text-charcoal"
   193→                        : "border-charcoal/20 bg-white text-charcoal/40"
   194→                    }`}
   195→                  >
   196→                    {i < step ? <Check className="h-3.5 w-3.5" /> : i + 1}
   197→                  </div>
   198→                  <span
   199→                    className={`hidden font-display text-xs uppercase tracking-wider sm:block ${
   200→                      i <= step ? "text-charcoal" : "text-charcoal/40"
   201→                    }`}
   202→                  >
   203→                    {label}
   204→                  </span>
   205→                  {i < STEPS.length - 1 && (
   206→                    <div
   207→                      className={`h-px flex-1 ${i < step ? "bg-teal" : "bg-charcoal/15"}`}
   208→                    />
   209→                  )}
   210→                </div>
   211→              ))}
   212→            </div>
   213→
   214→            {/* Body */}
   215→            <div className="scrollbar-thin flex-1 overflow-y-auto bg-ivory-deep/30 px-5 py-5 sm:px-6">
   216→              {/* STEP 0 — dates + room */}
   217→              {step === 0 && (
   218→                <div className="space-y-5">
   219→                  <div className="grid gap-5 md:grid-cols-2">
   220→                    {/* Calendar */}
   221→                    <div>
   222→                      <Label className="mb-2 flex items-center gap-1.5 font-display text-xs uppercase tracking-wider text-charcoal-soft">
   223→                        <Calendar className="h-3.5 w-3.5" /> Select check-in &amp; check-out
   224→                      </Label>
   225→                      <div className="rounded-2xl border border-charcoal/10 bg-white p-3">
   226→                        <CalendarPicker
   227→                          mode="range"
   228→                          numberOfMonths={1}
   229→                          disabled={(d) => d < new Date(new Date().setHours(0, 0, 0, 0))}
   230→                          selected={{
   231→                            from: dateRange.from,
   232→                            to: dateRange.to,
   233→                          }}
   234→                          onSelect={(range) =>
   235→                            setDateRange({ from: range?.from, to: range?.to })
   236→                          }
   237→                          className="rk-calendar"
   238→                        />
   239→                      </div>
   240→                    </div>
   241→
   242→                    {/* Room selection */}
   243→                    <div>
   244→                      <Label className="mb-2 flex items-center gap-1.5 font-display text-xs uppercase tracking-wider text-charcoal-soft">
   245→                        <Sparkles className="h-3.5 w-3.5" /> Choose your room
   246→                      </Label>
   247→                      <div className="scrollbar-thin max-h-[336px] space-y-2 overflow-y-auto pr-1">
   248→                        {rooms.length === 0 ? (
   249→                          <div className="flex h-32 items-center justify-center">
   250→                            <Loader2 className="h-5 w-5 animate-spin text-teal" />
   251→                          </div>
   252→                        ) : (
   253→                          rooms.map((r) => {
   254→                            const sel = r.id === selectedRoomId;
   255→                            return (
   256→                              <button
   257→                                key={r.id}
   258→                                onClick={() => setSelectedRoomId(r.id)}
   259→                                className={`flex w-full items-center gap-3 rounded-xl border p-2.5 text-left transition-all focus-ring ${
   260→                                  sel
   261→                                    ? "border-gold bg-gold/10 shadow-sm"
   262→                                    : "border-charcoal/10 bg-white hover:border-teal/30"
   263→                                }`}
   264→                                aria-pressed={sel}
   265→                              >
   266→                                <div
   267→                                  className="h-14 w-14 shrink-0 rounded-lg bg-cover bg-center"
   268→                                  style={{
   269→                                    backgroundImage: `url(${JSON.parse(r.imageUrls || "[]")[0] || "/images/heritage-room.jpg"})`,
   270→                                  }}
   271→                                />
   272→                                <div className="min-w-0 flex-1">
   273→                                  <div className="font-serif text-sm font-semibold text-charcoal">
   274→                                    {r.name}
   275→                                  </div>
   276→                                  <div className="font-display text-xs text-charcoal-soft">
   277→                                    {r.maxGuests} guests · {r.sizeSqft} sq.ft · {r.bedType}
   278→                                  </div>
   279→                                  <div className="mt-0.5 font-serif text-sm font-bold text-teal">
   280→                                    ₹{r.basePrice.toLocaleString("en-IN")}
   281→                                    <span className="font-sans text-[10px] font-normal text-charcoal-soft">
   282→                                      {" "}/ night
   283→                                    </span>
   284→                                  </div>
   285→                                </div>
   286→                                <div
   287→                                  className={`grid h-5 w-5 shrink-0 place-items-center rounded-full border-2 ${
   288→                                    sel ? "border-gold bg-gold" : "border-charcoal/20"
   289→                                  }`}
   290→                                >
   291→                                  {sel && <Check className="h-3 w-3 text-charcoal" />}
   292→                                </div>
   293→                              </button>
   294→                            );
   295→                          })
   296→                        )}
   297→                      </div>
   298→                    </div>
   299→                  </div>
   300→
   301→                  {/* Live price preview */}
   302→                  {pricing && (
   303→                    <div className="rounded-2xl border border-gold/30 bg-white p-4">
   304→                      <div className="mb-2 flex items-center justify-between">
   305→                        <span className="font-serif text-sm font-semibold text-charcoal">
   306→                          Live price estimate
   307→                        </span>
   308→                        <span className="rounded-full bg-teal/8 px-2 py-0.5 font-display text-[10px] uppercase tracking-wider text-teal">
   309→                          {nights} {nights === 1 ? "night" : "nights"}
   310→                        </span>
   311→                      </div>
   312→                      <div className="space-y-1.5 font-display text-sm text-charcoal-soft">
   313→                        <Row label={`₹${pricing.perNight.toLocaleString("en-IN")} × ${nights}`} value={`₹${pricing.subtotal.toLocaleString("en-IN")}`} />
   314→                        <Row label="GST (12%)" value={`₹${pricing.taxes.toLocaleString("en-IN")}`} />
   315→                        <Row label="Service fee" value={`₹${pricing.serviceFee.toLocaleString("en-IN")}`} />
   316→                        <div className="mt-2 flex items-center justify-between border-t border-charcoal/10 pt-2">
   317→                          <span className="font-serif text-base font-semibold text-charcoal">Total</span>
   318→                          <span className="font-serif text-xl font-bold text-teal">
   319→                            ₹{pricing.total.toLocaleString("en-IN")}
   320→                          </span>
   321→                        </div>
   322→                      </div>
   323→                    </div>
   324→                  )}
   325→                </div>
   326→              )}
   327→
   328→              {/* STEP 1 — guests + contact */}
   329→              {step === 1 && (
   330→                <div className="space-y-5">
   331→                  <div>
   332→                    <Label className="mb-2 flex items-center gap-1.5 font-display text-xs uppercase tracking-wider text-charcoal-soft">
   333→                      <Users className="h-3.5 w-3.5" /> Number of guests
   334→                    </Label>
   335→                    <div className="grid grid-cols-2 gap-3">
   336→                      <GuestCounter label="Adults" value={guests.adults} min={1} max={selectedRoom?.maxGuests || 6} onChange={(v) => setGuests((g) => ({ ...g, adults: v }))} />
   337→                      <GuestCounter label="Children" value={guests.children} min={0} max={4} onChange={(v) => setGuests((g) => ({ ...g, children: v }))} />
   338→                    </div>
   339→                    {selectedRoom && (
   340→                      <p className="mt-2 font-display text-xs text-charcoal-soft">
   341→                        Max {selectedRoom.maxGuests} guests for {selectedRoom.name}.
   342→                      </p>
   343→                    )}
   344→                  </div>
   345→                  <div className="grid gap-3 sm:grid-cols-2">
   346→                    <div>
   347→                      <Label htmlFor="bk-name" className="mb-1.5 flex items-center gap-1.5 font-display text-xs uppercase tracking-wider text-charcoal-soft">
   348→                        <User className="h-3.5 w-3.5" /> Full name
   349→                      </Label>
   350→                      <Input id="bk-name" value={contact.name} onChange={(e) => setContact((c) => ({ ...c, name: e.target.value }))} placeholder="Anjali Mehta" className="bg-white" />
   351→                    </div>
   352→                    <div>
   353→                      <Label htmlFor="bk-email" className="mb-1.5 flex items-center gap-1.5 font-display text-xs uppercase tracking-wider text-charcoal-soft">
   354→                        <Mail className="h-3.5 w-3.5" /> Email
   355→                      </Label>
   356→                      <Input id="bk-email" type="email" value={contact.email} onChange={(e) => setContact((c) => ({ ...c, email: e.target.value }))} placeholder="anjali@example.com" className="bg-white" />
   357→                    </div>
   358→                    <div>
   359→                      <Label htmlFor="bk-phone" className="mb-1.5 flex items-center gap-1.5 font-display text-xs uppercase tracking-wider text-charcoal-soft">
   360→                        <Phone className="h-3.5 w-3.5" /> Phone (with country code)
   361→                      </Label>
   362→                      <Input id="bk-phone" value={contact.phone} onChange={(e) => setContact((c) => ({ ...c, phone: e.target.value }))} placeholder="+91 98765 43210" className="bg-white" />
   363→                    </div>
   364→                    <div>
   365→                      <Label htmlFor="bk-requests" className="mb-1.5 flex items-center gap-1.5 font-display text-xs uppercase tracking-wider text-charcoal-soft">
   366→                        <MessageSquare className="h-3.5 w-3.5" /> Special requests (optional)
   367→                      </Label>
   368→                      <Input id="bk-requests" value={contact.requests} onChange={(e) => setContact((c) => ({ ...c, requests: e.target.value }))} placeholder="Early check-in, airport pickup…" className="bg-white" />
   369→                    </div>
   370→                  </div>
   371→                  <div className="rounded-xl border border-gold/30 bg-gold/5 p-3">
   372→                    <div className="flex items-start gap-2">
   373→                      <ShieldCheck className="mt-0.5 h-4 w-4 shrink-0 text-teal" />
   374→                      <p className="font-display text-xs leading-relaxed text-charcoal-soft">
   375→                        Your details are encrypted in transit. We use them only to
   376→                        confirm your booking and send a calendar invite — never for
   377→                        third-party marketing.
   378→                      </p>
   379→                    </div>
   380→                  </div>
   381→                </div>
   382→              )}
   383→
   384→              {/* STEP 2 — payment */}
   385→              {step === 2 && (
   386→                <div className="space-y-5">
   387→                  <div>
   388→                    <Label className="mb-2 flex items-center gap-1.5 font-display text-xs uppercase tracking-wider text-charcoal-soft">
   389→                      <ShieldCheck className="h-3.5 w-3.5" /> Payment method
   390→                    </Label>
   391→                    <div className="grid gap-2 sm:grid-cols-3">
   392→                      {[
   393→                        { id: "RAZORPAY" as const, label: "Razorpay", sub: "UPI · Cards · Netbanking", flag: "🇮🇳" },
   394→                        { id: "STRIPE" as const, label: "Stripe", sub: "International cards", flag: "🌍" },
   395→                        { id: "PAY_AT_HOTEL" as const, label: "Pay at hotel", sub: "Reserve now, pay on arrival", flag: "🏨" },
   396→                      ].map((p) => (
   397→                        <button
   398→                          key={p.id}
   399→                          onClick={() => setPaymentMethod(p.id)}
   400→                          className={`rounded-xl border p-3 text-left transition-all focus-ring ${
   401→                            paymentMethod === p.id
   402→                              ? "border-gold bg-gold/10"
   403→                              : "border-charcoal/10 bg-white hover:border-teal/30"
   404→                          }`}
   405→                          aria-pressed={paymentMethod === p.id}
   406→                        >
   407→                          <div className="text-2xl">{p.flag}</div>
   408→                          <div className="mt-1 font-serif text-sm font-semibold text-charcoal">{p.label}</div>
   409→                          <div className="font-display text-[10px] text-charcoal-soft">{p.sub}</div>
   410→                        </button>
   411→                      ))}
   412→                    </div>
   413→                  </div>
   414→
   415→                  {/* Final price summary */}
   416→                  {pricing && selectedRoom && dateRange.from && dateRange.to && (
   417→                    <div className="rounded-2xl border border-teal/30 bg-white p-4">
   418→                      <div className="mb-3 font-serif text-sm font-semibold text-charcoal">
   419→                        Booking summary
   420→                      </div>
   421→                      <div className="mb-3 flex items-center gap-3 border-b border-charcoal/10 pb-3">
   422→                        <div
   423→                          className="h-14 w-14 shrink-0 rounded-lg bg-cover bg-center"
   424→                          style={{
   425→                            backgroundImage: `url(${JSON.parse(selectedRoom.imageUrls || "[]")[0] || "/images/heritage-room.jpg"})`,
   426→                          }}
   427→                        />
   428→                        <div>
   429→                          <div className="font-serif text-sm font-semibold text-charcoal">
   430→                            {selectedRoom.name}
   431→                          </div>
   432→                          <div className="font-display text-xs text-charcoal-soft">
   433→                            {dateRange.from.toLocaleDateString("en-IN", { day: "numeric", month: "short" })}
   434→                            {" → "}
   435→                            {dateRange.to.toLocaleDateString("en-IN", { day: "numeric", month: "short" })}
   436→                            {" · "}
   437→                            {nights} {nights === 1 ? "night" : "nights"}
   438→                          </div>
   439→                          <div className="font-display text-xs text-charcoal-soft">
   440→                            {guests.adults} adults{guests.children > 0 ? ` · ${guests.children} children` : ""}
   441→                          </div>

   443→                      </div>
   444→                      <div className="space-y-1.5 font-display text-sm text-charcoal-soft">

   446→                        <Row label="GST (12%)" value={`₹${pricing.taxes.toLocaleString("en-IN")}`} />
   447→                        <Row label="Service fee" value={`₹${pricing.serviceFee.toLocaleString("en-IN")}`} />

   457→
   458→                  <div className="flex items-start gap-2 rounded-xl border border-marsala/20 bg-marsala/5 p-3">
   459→                    <ShieldCheck className="mt-0.5 h-4 w-4 shrink-0 text-marsala" />

   465→                  </div>

   468→
   469→              {/* STEP 3 — confirmation */}

   582→              {step === 2 && (

   607→                </Button>

   609→            </div>
   610→          </motion.div>
   611→        </motion.div>
   612→      )}
   613→    </AnimatePresence>
   614→  );
   615→}

   626→function GuestCounter({
   627→  label, value, min, max, onChange,
   628→}: {
   629→  label: string; value: number; min: number; max: number; onChange: (v: number) => void;
   630→}) {
   631→  return (
   632→    <div className="flex items-center justify-between rounded-xl border border-charcoal/10 bg-white p-3">

   634→      <div className="flex items-center gap-3">
   635→        <button
   636→          onClick={() => onChange(Math.max(min, value - 1))}
   637→          className="grid h-8 w-8 place-items-center rounded-full border border-charcoal/15 text-charcoal transition-colors hover:border-teal hover:bg-teal hover:text-ivory disabled:opacity-40 focus-ring"

   646→          className="grid h-8 w-8 place-items-center rounded-full border border-charcoal/15 text-charcoal transition-colors hover:border-teal hover:bg-teal hover:text-ivory disabled:opacity-40 focus-ring"

   648→          aria-label={`Increase ${label}`}
   649→        >
   650→          +
   651→        </button>
   652→      </div>
   653→    </div>
   654→  );
