   146→   146→function AdminLogin({ onLogin, onBack }: { onLogin: (a: AdminUser, token: string) => void; onBack: () => void }) {
   147→   147→  const [email, setEmail] = useState("admin@rkresidency.in");
   148→   148→  const [password, setPassword] = useState("rk-admin-2026");
   149→   149→  const [loading, setLoading] = useState(false);
   150→   150→  const [error, setError] = useState("");
   151→   151→
   152→   152→  const submit = async (e: React.FormEvent) => {
   153→   153→    e.preventDefault();
   154→   154→    setLoading(true);
   155→   155→    setError("");
   156→   156→    for (let attempt = 0; attempt < 5; attempt++) {
   157→   157→      try {
   158→   158→        const res = await fetch("/api/admin/auth", {
   159→   159→          method: "POST",
   160→   160→          headers: { "Content-Type": "application/json" },
   161→   161→          body: JSON.stringify({ email, password }),
   162→   162→        });
   163→   163→
   164→   164→        // Check if response is OK
   165→   165→        if (!res.ok) {
   166→   166→          // Try to parse error JSON, but handle non-JSON responses
   167→   167→          let errMsg = `Server error (${res.status})`;
   168→   168→          try {
   169→   169→            const errData = await res.json();
   170→   170→            errMsg = errData.error || errMsg;
   171→   171→          } catch {}
   172→   172→          throw new Error(errMsg);
   173→   173→        }
   174→   174→
   175→   175→        // Parse JSON response
   176→   176→        const d = await res.json();
   177→   177→        if (!d.ok || !d.admin || !d.token) {
   178→   178→          throw new Error(d.error || "Invalid response from server");
   179→   179→        }
   180→   180→
   181→   181→        // Success!
   182→   182→        onLogin(d.admin, d.token);
   183→   183→        toast.success(`Welcome, ${d.admin.name}`);
   184→   184→        setLoading(false);
   185→   185→        return;
   186→   186→      } catch (e: unknown) {
   187→   187→        const msg = e instanceof Error ? e.message : "Login failed";
   188→   188→
   189→   189→        // Retry on network errors, timeouts, or server errors
   190→   190→        const shouldRetry = (
   191→   191→          msg.includes("fetch") ||
   192→   192→          msg.includes("network") ||
   193→   193→          msg.includes("timeout") ||
   194→   194→          msg.includes("500") ||
   195→   195→          msg.includes("502") ||
   196→   196→          msg.includes("503") ||
   197→   197→          msg.includes("Load failed")
   198→   198→        );
   199→   199→
   200→   200→        if (shouldRetry && attempt < 4) {
   201→   201→          setError(`Connecting to server... retry ${attempt + 2}/5`);
   202→   202→          await new Promise((r) => setTimeout(r, 1500));
   203→   203→          continue;
   204→   204→        }
   205→   205→
   206→   206→        // Final failure
   207→   207→        setError(msg);
   208→   208→        toast.error("Login failed", { description: msg });
   209→   209→        setLoading(false);
   210→   210→        return;
   211→   211→      }
   212→   212→    }
   213→   213→    setError("Unable to connect. Please refresh the page and try again.");
   214→   214→    setLoading(false);
   215→   215→  };
   216→   216→
   217→   217→  return (
   218→   218→    <div className="grid min-h-screen place-items-center bg-charcoal p-4">
   219→   219→      <div className="pointer-events-none absolute inset-0 bg-yamuna-ripple opacity-15" />
   220→   220→      <motion.div initial={{ opacity: 0, scale: 0.96 }} animate={{ opacity: 1, scale: 1 }} className="relative w-full max-w-md overflow-hidden rounded-3xl bg-ivory shadow-2xl">
   221→   221→        <div className="border-b border-charcoal/10 bg-teal px-6 py-5 text-ivory">
   222→   222→          <div className="flex items-center gap-3">
   223→   223→            <span className="grid h-11 w-11 place-items-center rounded-full border border-gold/40 bg-ivory/5 text-gold"><Logo size={24} /></span>
   224→   224→            <div><div className="font-serif text-lg font-semibold">RK Residency Admin</div><div className="font-display text-[10px] uppercase tracking-[0.28em] text-gold-soft">Restricted access</div></div>
   225→   225→          </div>
   226→   226→        </div>
   227→   227→        <form onSubmit={submit} className="space-y-4 p-6">
   228→   228→          <div><Label htmlFor="a-email" className="mb-1.5 block font-display text-xs uppercase tracking-wider text-charcoal-soft">Email</Label><Input id="a-email" type="email" value={email} onChange={(e) => setEmail(e.target.value)} required className="bg-ivory-deep/40" /></div>
   229→   229→          <div><Label htmlFor="a-pass" className="mb-1.5 block font-display text-xs uppercase tracking-wider text-charcoal-soft">Password</Label><div className="relative"><Input id="a-pass" type="password" value={password} onChange={(e) => setPassword(e.target.value)} required className="bg-ivory-deep/40" /><button type="button" onClick={() => {}} className="absolute right-3 top-1/2 -translate-y-1/2 text-charcoal-soft">👁️</button></div></div>
   230→   230→          {error && <div className="rounded-xl border border-marsala/30 bg-marsala/5 px-3 py-2 font-display text-xs text-marsala">{error}</div>}
   231→   231→          <Button type="submit" disabled={loading} className="w-full rounded-full bg-teal py-3 font-serif text-base font-semibold text-ivory hover:bg-teal-deep disabled:opacity-50">{loading ? <><Loader2 className="mr-2 h-4 w-4 animate-spin" /> Signing in…</> : <><Lock className="mr-2 h-4 w-4" /> Sign in</>}</Button>
   232→   232→          <div className="rounded-xl border border-gold/30 bg-gold/5 px-3 py-2 font-display text-[11px] leading-relaxed text-charcoal-soft"><strong className="text-gold-deep">Demo credentials</strong> — admin@rkresidency.in / rk-admin-2026</div>
   233→   233→          <button type="button" onClick={onBack} className="block w-full text-center font-display text-xs text-charcoal-soft hover:text-charcoal">← Back to website</button>
   234→   234→        </form>
   235→   235→      </motion.div>
   236→   236→    </div>
   237→   237→  );
   238→   238→}
